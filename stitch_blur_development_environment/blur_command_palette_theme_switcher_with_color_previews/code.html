// dispatch over bounded ring buffers with deterministic backpressure.

pub struct ProtocolHandler<T: Stream> {

transport: Arc<T>,

inbound_tx: mpsc::Sender<Message>,

max_capacity: usize,

}

impl<T> ProtocolHandler<T> where T: Stream + Send + Sync + 'static {

pub fn new(transport: T, capacity: usize) -> Self {

let (tx, _rx) = mpsc::channel(capacity);

Self {

transport: Arc::new(transport),

max_capacity: capacity,

}

}
